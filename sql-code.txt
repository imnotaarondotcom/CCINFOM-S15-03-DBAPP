DROP DATABASE IF EXISTS movie_ticket_management;

CREATE DATABASE IF NOT EXISTS movie_ticket_management;
USE movie_ticket_management;

CREATE TABLE Movies(
movie_id INT PRIMARY KEY AUTO_INCREMENT,
movie_name VARCHAR(45),
genre VARCHAR(45),
age_rating ENUM('G','PG','PG-13','R'),
duration INT
);

CREATE TABLE Customers(
customer_id INT PRIMARY KEY AUTO_INCREMENT,
phone_number VARCHAR(11),
username VARCHAR(45), 
account_type ENUM('Admin','Customer'),
user_password VARCHAR(255)
);

CREATE TABLE Venues(
venue_id INT PRIMARY KEY AUTO_INCREMENT,
venue_name VARCHAR(45),
address VARCHAR(100)
);

CREATE TABLE Rooms(
room_id INT PRIMARY KEY AUTO_INCREMENT,
room_name VARCHAR(45),
room_type ENUM('Standard','IMAX','VIP'),
venue_id INT,
FOREIGN KEY(venue_id) REFERENCES Venues(venue_id)
);

CREATE TABLE Seats(
seat_id INT PRIMARY KEY AUTO_INCREMENT,
seat_no int,
room_id int,
FOREIGN KEY(room_id) REFERENCES Rooms(room_id)
);

CREATE TABLE Screenings(
screening_id INT PRIMARY KEY AUTO_INCREMENT,
movie_id INT,
venue_id INT,
room_id INT,
price DECIMAL(8,2),
screening_date DATE,
screening_start_time TIME,
screening_end_time TIME,
screening_status VARCHAR(20) DEFAULT 'Active',
FOREIGN KEY(movie_id) REFERENCES Movies(movie_id),
FOREIGN KEY(venue_id) REFERENCES Venues(venue_id),
FOREIGN KEY(room_id) REFERENCES Rooms(room_id)
);

CREATE TABLE TicketBookings(
ticket_no INT PRIMARY KEY AUTO_INCREMENT,
ticket_status ENUM('Booked','Cancelled','Refunded'),
seat_id INT,
date_booked DATETIME,
customer_id INT,
screening_id INT,
FOREIGN KEY(customer_id) REFERENCES Customers(customer_id),
FOREIGN KEY(screening_id) REFERENCES Screenings(screening_id),
FOREIGN KEY(seat_id) REFERENCES Seats(seat_id)
);
